function renderGroupedMaterials() { const c = document.getElementById('materials-container'); if (!c) return; const role = String(currentUser.role || '').trim().toUpperCase(); const materials = appData.materials || []; // JIKA GURU / ADMIN: Dikelompokkan berdasarkan Mapel if (role === 'GURU' || role === 'ADMIN') { const subjects = appData.subjects || []; if (subjects.length === 0) { c.innerHTML = '
Tidak ada mata pelajaran yang tersedia.
'; return; } c.innerHTML = subjects.map((subj, idx) => { const subjMaterials = materials.filter(m => String(m.subject_id) === String(subj.id)); const groupId = `mat-group-${idx}`; return `
📘 MAPEL: ${escapeHtml(subj.subject_name)} (${subjMaterials.length}) ▼
${subjMaterials.length === 0 ? '

Belum ada materi untuk mata pelajaran ini.

' : subjMaterials.map(m => `

${escapeHtml(m.title)}

${m.url ? `BUKA TAUTAN MATERI 🔗` : ''}
`).join('')}
`; }).join(''); } // JIKA SISWA: Dikelompokkan berdasarkan Kelas yang diikuti else { const classes = appData.classes || []; if (classes.length === 0) { c.innerHTML = '
Tidak ada kelas yang diikuti.
'; return; } c.innerHTML = classes.map((cls, idx) => { const clsMaterials = materials.filter(m => { const raw = String(m.class_id || '').trim(); if (!raw || raw === 'ALL') return true; return raw.split(',').map(x => x.trim()).includes(String(cls.id)); }); const groupId = `mat-class-group-${idx}`; return `
🏫 KELAS: ${escapeHtml(cls.class_name)} (${clsMaterials.length}) ▼
${clsMaterials.length === 0 ? '

Belum ada materi untuk kelas ini.

' : clsMaterials.map(m => `
